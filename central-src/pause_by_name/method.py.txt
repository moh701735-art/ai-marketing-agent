    async def _pause_by_name(self, ctx: str, params: dict, idem) -> tuple:
        """p240a: «طفي حملة X». يقرأ حملات الحسابات (read_campaigns بصفحات)، يطابق الاسم بالضبط (تطبيع عربي)، يطفي الشغّال فقط بـset_campaign_status DISABLE،
        ثم يقرأ الحالة من تيك توك للتأكد. الصلاحيات/الـidempotency/السجل تمر عبر self.call لكل خطوة. لا يشغّل أي حملة أبداً."""
        def bad(msg):
            return 400, {"ok": False, "error": {"class": "invalid_params", "message": msg}}
        name, accts = params.get("name"), params.get("accounts")
        if set(params) - {"name", "accounts"} or not isinstance(name, str) or not (1 <= len(name.strip()) <= 80):
            return bad("name (1..80)")
        if not isinstance(accts, list) or not (1 <= len(accts) <= 40) or not all(isinstance(a, str) and __import__("re").fullmatch(r"[0-9]{5,24}", a) for a in accts):
            return bad("accounts: list of 1..40 ad account ids")
        if not isinstance(idem, str) or not (16 <= len(idem) <= 128):
            return bad("idempotency_key (16..128 chars) required")

        def norm(s):
            import re as _re, unicodedata as _ud
            s = _ud.normalize("NFKC", str(s or "")).lower()
            s = "".join(ch for ch in s if _ud.category(ch) != "Mn")                     # تشكيل
            for a, b in (("أ", "ا"), ("إ", "ا"), ("آ", "ا"), ("ة", "ه"), ("ى", "ي"), ("ـ", "")):
                s = s.replace(a, b)
            s = s.translate({0x0660 + i: ord(str(i)) for i in range(10)})                  # أرقام عربية
            return _re.sub(r"\s+", " ", _re.sub(r"[^\w\s]", " ", s)).strip()
        want = norm(name)
        if not want:
            return bad("name: no searchable words")
        sem = asyncio.Semaphore(8); camps = {}; accounts_failed = {}

        def why(env):
            try:
                e = (env or {}).get("error") or {}
                e = e if isinstance(e, dict) else {"message": str(e)}
                return " ".join(p for p in (str(e.get("class") or ""), str(e.get("message") or "")[:90]) if p).strip() or "no_response"
            except Exception:
                return "error"

        async def scan(acct):
            """كل حملات الحساب: [(campaign_id, name, operation_status)]، أو None عند الفشل."""
            out = []; page = 1
            async with sem:
                while page <= 10:
                    try:
                        st, env = await self._rc(ctx, "tiktok", "tiktok.read_campaigns", acct, {"page": page, "page_size": 100})
                    except Exception as ex:
                        accounts_failed[acct] = type(ex).__name__ + ": " + str(ex)[:70]
                        return None
                    if st != 200 or not (env or {}).get("ok"):
                        accounts_failed[acct] = why(env)
                        return None
                    d = ((env.get("data") or {}).get("data") or {})
                    for c in (d.get("list") or []):
                        if c.get("campaign_id") and c.get("campaign_name"):
                            out.append((str(c["campaign_id"]), str(c["campaign_name"]), str(c.get("operation_status") or "")))
                    tp = int(((d.get("page_info") or {}).get("total_page")) or 1)
                    if page >= tp:
                        break
                    page += 1
            return out

        for acct, lst in zip(accts, await asyncio.gather(*[scan(a) for a in accts])):
            if lst is not None:
                camps[acct] = lst
        live = [(a, cid, nm, stt) for a, lst in camps.items() for cid, nm, stt in lst if stt.upper() != "DELETE"]
        exact = [x for x in live if norm(x[2]) == want]
        mode = "exact"
        if not exact:
            contains = [x for x in live if want in norm(x[2]) and x[3].upper() == "ENABLE"]
            if len(contains) > 1:
                return 200, {"ok": True, "data": {"mode": "ambiguous", "ambiguous": [{"account": a, "campaign_id": c, "name": n} for a, c, n, _ in contains[:10]], "matched": 0,
                                                   "paused": [], "already_paused": [], "failed": [], "accounts_failed": accounts_failed}}
            if len(contains) == 1:
                exact = contains; mode = "contains"
        if not exact:
            wt = set(w for w in want.split() if len(w) >= 2)
            sim = sorted(((len(wt & set(norm(n).split())), n) for _, _, n, _ in live if wt & set(norm(n).split())), reverse=True)
            names = []
            for _, n in sim:
                if n not in names:
                    names.append(n)
            return 200, {"ok": True, "data": {"mode": "not_found", "matched": 0, "paused": [], "already_paused": [], "failed": [], "similar": names[:5], "accounts_failed": accounts_failed}}
        running = [x for x in exact if x[3].upper() == "ENABLE"]
        already = [{"account": a, "campaign_id": c, "name": n} for a, c, n, s in exact if s.upper() != "ENABLE"]
        paused = []; failed = []

        async def stop(x):
            a, cid, nm, _ = x
            try:
                st, env = await self.call(ctx, "tiktok", "tiktok.set_campaign_status", a, {"campaign_ids": [cid], "operation_status": "DISABLE"}, (idem[:100] + "-" + cid)[:128])
            except Exception as ex:
                return {"account": a, "campaign_id": cid, "name": nm, "why": type(ex).__name__ + ": " + str(ex)[:70]}, False
            if st == 200 and (env or {}).get("ok"):
                return {"account": a, "campaign_id": cid, "name": nm}, True
            return {"account": a, "campaign_id": cid, "name": nm, "why": why(env)}, False

        for item, ok in await asyncio.gather(*[stop(x) for x in running]):
            (paused if ok else failed).append(item)
        for a in sorted({p["account"] for p in paused}):          # قراءة الحالة بعد الإيقاف (حساب واحد = قراءة واحدة)
            lst = await scan(a)
            now = {c: s for c, _, s in (lst or [])}
            for p in paused:
                if p["account"] == a:
                    p["verified"] = bool(lst is not None and now.get(p["campaign_id"], "").upper() == "DISABLE")
        return 200, {"ok": True, "data": {"mode": mode, "matched": len(exact), "paused": paused, "already_paused": already, "failed": failed, "accounts_failed": accounts_failed}}
