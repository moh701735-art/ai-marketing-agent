def sub1(s, pat, fn, what, kw=None, flags=0):
    r, n = re.subn(pat, fn, s, flags=flags)

        ex = [repr(l[:150]) for l in s.split('\n') if kw and kw in l][:3]
        fail('%s anchor count=%d; lines with %r: %s' % (what, n, kw, ' | '.join(ex) or 'none'))
    return r
def lit1(s, a, b, what):
