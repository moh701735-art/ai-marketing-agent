import os, re, shutil, sys
BASE = os.environ.get('P224_BASE', '/opt/central/')
P = 'broker/app/service.py'
s = open(BASE + P, encoding='utf-8').read()
if 'p224' in s: print('APPLY224_OK changed=0 (already applied)'); sys.exit(0)
r = re.compile(r'(?m)^([ \t]*)fname = \(p\["campaign_name"\]\[:100\] \+ "\.mp4"\)[^\n]*$')
m = list(r.finditer(s))
if len(m) != 1:
    sys.exit('PATCH FAIL (nothing written): fname anchor count=%d; lines: %s' % (len(m), ' | '.join(ascii(l.strip()[:200]) for l in s.split('\n') if 'fname =' in l)[:600] or 'none'))
I = m[0].group(1)
new_txt = (I + 'import hashlib as _h224, time as _t224   # p224\n' +
           I + 'fname = (p["campaign_name"][:85] + "-" + _h224.md5(("%s|%s" % (_t224.time(), idem)).encode()).hexdigest()[:6] + ".mp4")   # p224: اسم ملف فريد لكل رفع (تيك توك يرفض الاسم المكرر: Duplicated material name)')
new = s[:m[0].start()] + new_txt + s[m[0].end():]
try: compile(new, P, 'exec')
except SyntaxError as e: sys.exit('PATCH FAIL (nothing written): syntax %s' % e)
shutil.copy(BASE + P, BASE + P + '.pre224'); open(BASE + P, 'w', encoding='utf-8').write(new)
print('APPLY224_OK TikTok video uploads get a unique file name (fixes Duplicated material name)')
